import { defineCollection, reference, z } from 'astro:content';
import { glob } from 'astro/loaders';
import { TIPOS, MOVIMIENTOS, TEMAS, MOTIVOS } from './utils/taxonomia';

// Decap stores cleared optional fields inconsistently:
//   - cleared text: ""
//   - cleared multi-select (lists): null
//   - cleared empty list: [] (rare, but observed)
// These helpers normalize those values to `undefined` so the schema validates
// as the user intended (field absent, not field invalid).

const optionalDate = () =>
  z.preprocess(
    (v) => (v === '' || v === null ? undefined : v),
    z.coerce.date().optional(),
  );

const optionalString = () =>
  z.preprocess(
    (v) => (v === '' || v === null ? undefined : v),
    z.string().optional(),
  );

const optionalUrl = () =>
  z.preprocess(
    (v) => (v === '' || v === null ? undefined : v),
    z.string().url().optional(),
  );

const optionalInt = () =>
  z.preprocess(
    (v) => (v === '' || v === null ? undefined : v),
    z.coerce.number().int().optional(),
  );

const poemas = defineCollection({
  loader: glob({ pattern: '**/*.md', base: './src/content/poemas' }),
  schema: ({ image }) =>
    z.object({
      titulo: z.string(),
      resumen: optionalString(),
      fecha: z.coerce.date(),
      fecha_actualizada: optionalDate(),
      ilustracion: z.optional(image()),
      spotify_url: optionalUrl(),
      borrador: z.boolean().default(false),
      tipo: z.enum(TIPOS),
      movimiento: z.preprocess(
        (v) => (v === '' || v === null ? undefined : v),
        z.enum(MOVIMIENTOS).optional(),
      ),
      temas: z.preprocess(
        (v) => (v === null || v === '' ? [] : v),
        z.array(z.enum(TEMAS)).min(1),
      ),
      motivos: z.preprocess(
        (v) => (v === null || v === '' ? undefined : v),
        z.array(z.enum(MOTIVOS)).default([]),
      ),
      autor: reference('autores'),
      traductor: z.preprocess(
        (v) => (v === '' || v === null ? undefined : v),
        reference('autores').optional(),
      ),
      curador: z.string().default('Don Alejandro'),
      es_seudonimo: z.boolean().default(true),
      nombre_real: optionalString(),
      nota_curador: optionalString(),
    }),
});

const autores = defineCollection({
  loader: glob({ pattern: '**/*.md', base: './src/content/autores' }),
  schema: ({ image }) =>
    z.object({
      nombre: z.string(),
      tipo: z.enum(['clasico', 'colaborador']),
      descripcion: optionalString(),
      nacionalidad: optionalString(),
      nacimiento: optionalInt(),
      muerte: optionalInt(),
      lugar_nacimiento: optionalString(),
      lugar_muerte: optionalString(),
      imagen: z.optional(image()),
      sameAs: z.preprocess(
        (v) => {
          if (v === null || v === '' || v === undefined) return undefined;
          if (Array.isArray(v) && v.length === 0) return undefined;
          return v;
        },
        z.array(z.string().url()).optional(),
      ),
      tambien_en: optionalString(),
      github_username: optionalString(),
      nombre_real: optionalString(),
    }),
});

const entradas = defineCollection({
  loader: glob({ pattern: '**/*.md', base: './src/content/entradas' }),
  schema: ({ image }) =>
    z
      .object({
        titulo: z.string(),
        resumen: optionalString(),
        fecha: z.coerce.date(),
        fecha_actualizada: optionalDate(),
        ilustracion: z.optional(image()),
        spotify_url: optionalUrl(),
        borrador: z.boolean().default(false),
        curador: z.string().default('Don Alejandro'),
        es_seudonimo: z.boolean().default(true),
        nombre_real: optionalString(),
        tipo: z.enum(['bitacora', 'correspondencia']).default('bitacora'),
        remitente: optionalString(),
        poema_referido: z.preprocess(
          (v) => (v === '' || v === null ? undefined : v),
          reference('poemas').optional(),
        ),
      })
      .refine(
        (data) =>
          data.tipo !== 'correspondencia' ||
          (typeof data.remitente === 'string' && data.remitente.length > 0),
        {
          message:
            "Las entradas con tipo='correspondencia' requieren `remitente`. Usa 'Un lector' para cartas anónimas.",
          path: ['remitente'],
        },
      ),
});

const destacado = defineCollection({
  loader: glob({ pattern: '*.md', base: './src/content/destacado' }),
  schema: z
    .object({
      activo: z.boolean().default(true),
      cintillo: z.string(),
      tipo: z.enum(['poema', 'entrada', 'autor']),
      poema_referido: z.preprocess(
        (v) => (v === '' || v === null ? undefined : v),
        reference('poemas').optional(),
      ),
      entrada_referida: z.preprocess(
        (v) => (v === '' || v === null ? undefined : v),
        reference('entradas').optional(),
      ),
      autor_referido: z.preprocess(
        (v) => (v === '' || v === null ? undefined : v),
        reference('autores').optional(),
      ),
    })
    .refine(
      (data) => {
        if (!data.activo) return true;
        if (data.tipo === 'poema') return data.poema_referido !== undefined;
        if (data.tipo === 'entrada') return data.entrada_referida !== undefined;
        if (data.tipo === 'autor') return data.autor_referido !== undefined;
        return false;
      },
      {
        message:
          'Cuando `activo: true`, debes seleccionar el elemento referido del tipo correspondiente.',
        path: ['tipo'],
      },
    ),
});;

export const collections = { poemas, autores, entradas, destacado };

const autores = await getCollection('autores');
const poemas = await getCollection('poemas', ({ data }) => !data.borrador);

// Count published poems per author. `p.data.autor` is a reference()
// ({ collection, id }) in El Poemario — NOT a string like in El Fabulario.
// Index by `.id` (the file slug) to get a stable string key.
const conteoPorId = new Map<string, number>();
for (const p of poemas) {
  const k = p.data.autor.id;
  conteoPorId.set(k, (conteoPorId.get(k) ?? 0) + 1);
}

// Enrich + sort: poem count desc, alphabetical by name as tiebreak (es locale).
// Mirrors El Fabulario's /autores/ ordering. Classics and collaborators are
// not separated visually; the rubric in each AutorCard surfaces tipo inline.
const autoresConConteo = autores
  .map((a) => ({
    ...a,
    conteo: conteoPorId.get(a.id) ?? 0,
  }))
  .sort((a, b) => {
    if (b.conteo !== a.conteo) return b.conteo - a.conteo;
    return a.data.nombre.localeCompare(b.data.nombre, 'es');
  });

<script>
	type TipoEntidad = 'autor' | 'forma' | 'movimiento' | 'tema' | 'motivo';

	interface Entidad {
		tipo: TipoEntidad;
		slug: string;
		nombre: string;
		conteo: number;
		url: string;
	}

	const ETIQUETAS_TIPO: Record<TipoEntidad, string> = {
		autor: 'Autor',
		forma: 'Forma',
		movimiento: 'Movimiento',
		tema: 'Tema',
		motivo: 'Motivo',
	};

	function norm(s: string): string {
		return s.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
	}

	function escape(s: string): string {
		return s.replace(/[&<>"']/g, (c) => ({
			'&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;',
		})[c] as string);
	}

	const params = new URLSearchParams(window.location.search);
	const q = (params.get('q') ?? '').trim();

	const titulo = document.querySelector<HTMLElement>('[data-titulo]')!;
	const subtitulo = document.querySelector<HTMLElement>('[data-subtitulo]')!;
	const secEnt = document.querySelector<HTMLElement>('[data-seccion-entidades]')!;
	const secPoe = document.querySelector<HTMLElement>('[data-seccion-poemas]')!;
	const listaEnt = document.querySelector<HTMLElement>('[data-lista-entidades]')!;
	const listaPoe = document.querySelector<HTMLElement>('[data-lista-poemas]')!;
	const conteoPoe = document.querySelector<HTMLElement>('[data-conteo-poemas]')!;
	const vacio = document.querySelector<HTMLElement>('[data-vacio]')!;

	if (q) {
		titulo.textContent = `Resultados para «${q}»`;
		subtitulo.hidden = true;
		ejecutar(q);
	}

	async function ejecutar(query: string) {
		// ── Entities ──
		try {
			const ents: Entidad[] = await fetch('/buscar/entidades.json').then((r) => r.json());
			const qn = norm(query);
			const matches = ents.filter((e) => norm(e.nombre).includes(qn));
			if (matches.length > 0) {
				listaEnt.innerHTML = matches
					.map(
						(e) =>
							`<li><a href="${e.url}">` +
								`<span class="rotulo">${ETIQUETAS_TIPO[e.tipo]}</span>` +
								`<span class="nombre">${escape(e.nombre)}</span>` +
								`<span class="conteo">${e.conteo === 1 ? '1 poema' : `${e.conteo} poemas`}</span>` +
							`</a></li>`,
					)
					.join('');
				secEnt.hidden = false;
			}
		} catch {
			/* sin entidades disponibles, seguimos sin esta sección */
		}

		// ── Text via Pagefind ──
		let poemas = 0;
		try {
			// URL built at runtime to keep Rollup out of build-time resolution.
			const url = '/pagefind/pagefind.js';
			const pf = await import(/* @vite-ignore */ url);
			await pf.init();
			const { results } = await pf.search(query);

			const filas: string[] = [];

			for (const r of results) {
				const d = await r.data();
				const urlPoema: string = d.url;
				const tituloPoema = d.meta?.title ?? urlPoema;
				const autor = d.meta?.author ?? null;
				const excerpt = d.excerpt ?? '';
				filas.push(
					`<li><a href="${urlPoema}">` +
						`<span class="t-titulo">${escape(tituloPoema)}</span>` +
						(autor ? `<span class="t-autor">· ${escape(autor)}</span>` : '') +
						(excerpt ? `<p class="t-excerpt">${excerpt}</p>` : '') +
					`</a></li>`,
				);
				poemas++;
			}

			if (poemas > 0) {
				listaPoe.innerHTML = filas.join('');
				conteoPoe.textContent = `(${poemas})`;
				secPoe.hidden = false;
			}
		} catch {
			/* Pagefind not available — entities only */
		}

		if (secEnt.hidden && secPoe.hidden) {
			vacio.hidden = false;
		}
	}
</script>

      }

        poemaError.textContent = '';

      function validarPoema(valor) {
        if (!valor) return { ok: true };
        if (!valor.includes('://')) return { ok: true };
        let url;
        try {
          url = new URL(valor);
        } catch (err) {
          return {
            ok: false,
            msg: 'El enlace no es una URL válida. Escribe solo el título, o pega una URL completa de elpoemario.com.',
          };
        }
        if (url.protocol !== 'http:' && url.protocol !== 'https:') {
          return {
            ok: false,
            msg: 'Solo se aceptan enlaces http o https.',
          };
        }
        if (!HOSTS_PERMITIDOS.includes(url.hostname)) {

            msg: 'Solo se aceptan enlaces de elpoemario.com. Si el poema está en otro sitio, escribe su título en su lugar.',

// Resolve referenced author entry.
const autor = await getEntry(autorRef);
if (!autor) {
  throw new Error(`Poema "${id}" referencia autor inexistente: ${autorRef.id}`);
}

// Resolve optional translator entry. Zod has already validated the
// reference exists; we still guard against an unexpected null return.
const traductor = traductorRef ? await getEntry(traductorRef) : null;
if (traductorRef && !traductor) {
  throw new Error(
    `Poema "${id}" referencia traductor inexistente: ${traductorRef.id}`,
  );
}

// JSON-LD for the Poem entity.
const site = Astro.site!;
const jsonLd = buildPoemaJsonLd(props as Props, autor, traductor, site);

// Author dates: "1836–1870" only if both years are known.
const fnac = autor.data.nacimiento;
const fmue = autor.data.muerte;
const anios = fnac && fmue
  ? `${fnac}–${fmue}`
  : null;

// Author nationality + dates line, e.g. "mexicano · 1849–1873".
const autorNacionalidad = autor.data.nacionalidad ?? null;
const autorSubmeta = [autorNacionalidad, anios].filter(Boolean).join(' · ');

// Curator note: block-level Markdown (paragraphs, lists, blockquotes).
// `breaks: true` honours single newlines as <br>, matching El Fabulario.
const notaCuradorHtml = nota_curador
  ? (marked.parse(nota_curador, { async: false, breaks: true }) as string)
  : null;

// Curator credit (decision 11).
const creditoCurador = notaCuradorHtml
  ? `Selección y nota de ${curador}.`
  : `Selección de ${curador}.`;

// Body class: verse vs prose-poetry.
const bodyClass = tipo === 'prosa-poetica' ? 'prose' : 'prose prose--verso';

// Description for <Base>.
const descripcion = resumen ?? `Poema de ${autor.data.nombre}, en El Poemario.`;

// ── Previous / next navigation ────────────────────────────────────────────
// Build a chronological-desc list of all non-draft poemas. The current
// entry is found by id and the neighbors are selected with circular
// wrapping at both ends. Replicates NavegacionFabulas behaviour.
const todos = await getCollection('poemas', ({ data }) => !data.borrador);
todos.sort((a, b) => b.data.fecha.getTime() - a.data.fecha.getTime());

const idx = todos.findIndex((p) => p.id === id);

// idx can legitimately be -1 when this layout renders a draft (drafts
// are excluded from `todos`). In that case neighbours collapse to the
// current entry, which makes the navigation harmless but visible.
const safeIdx = idx === -1 ? 0 : idx;
const total = todos.length;
const anteriorEntry = total > 0
  ? todos[(safeIdx - 1 + total) % total]
  : null;
const siguienteEntry = total > 0
  ? todos[(safeIdx + 1) % total]
  : null;

const anterior = anteriorEntry
  ? { slug: anteriorEntry.id, titulo: anteriorEntry.data.titulo }
  : null;
const siguiente = siguienteEntry
  ? { slug: siguienteEntry.id, titulo: siguienteEntry.data.titulo }
  : null;
